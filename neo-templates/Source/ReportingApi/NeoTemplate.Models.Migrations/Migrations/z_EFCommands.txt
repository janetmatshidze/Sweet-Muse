// TO GENERATE A MIGRATION
// =======================
// 'InitialMigration' is used as the name for the first migration
// For delta migrations, {Name} should be a short description of the feature/change
// Run the migration command in your solution root.
dotnet ef migrations add InitialMigration --project NeoTemplate.Models.Migrations --context ReportingDbContext

dotnet ef migrations add {Name} --project NeoTemplate.Models.Migrations --context ReportingDbContext

// TO ROLLBACK MIGRATIONS IN THE DB TO A SPECIFIC MIGRATION
// ========================================================
dotnet ef database update {PreviousMigrationName} --project NeoTemplate.Models.Migrations --context ReportingDbContext

// TO REMOVE THE LAST ADDED MIGRATION FROM CODE
// ============================================
dotnet ef migrations remove --project NeoTemplate.Models.Migrations --context ReportingDbContext
